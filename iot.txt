let idCounter = 101;

function registerExam(){

    // User Interface Layer
    let name = document.getElementById("studentName").value;
    let course = document.getElementById("courseCode").value;
    let payment = document.getElementById("payment").value;

    if(name.trim() === "" || course.trim() === ""){
        alert("Please fill all fields");
        return;
    }

    // Domain Object Layer Features
    let registration = {
        studentId : "STU" + idCounter,
        studentName : name,
        courseCode : course,
        status : "PENDING_PAYMENT"
    };

    idCounter++;

    // Technical Service Layer Features
    if(payment === "yes"){
        registration.status = "REGISTERED";
    }

    // Display Output
    document.getElementById("sid").innerText = registration.studentId;
    document.getElementById("sname").innerText = registration.studentName;
    document.getElementById("course").innerText = registration.courseCode;
    document.getElementById("status").innerText = registration.status;

    document.getElementById("result").style.display = "block";
}

</script>

<script>

let product = {

    productID : "P001",
    name : "Laptop",
    quantity : 50,
    price : 800.0

};

function updateStock(){

    let sold = parseInt(document.getElementById("sold").value);

    document.getElementById("successBox").style.display = "none";
    document.getElementById("errorBox").style.display = "none";

    if(isNaN(sold) || sold <= 0){

        alert("Please enter valid quantity");
        return;
    }

    // Technical Service Logic
    if(product.quantity - sold >= 0){

        product.quantity = product.quantity - sold;

        // Update UI
        document.getElementById("qty").innerText = product.quantity;
        document.getElementById("finalQty").innerText = product.quantity;

        document.getElementById("successBox").style.display = "block";

    }else{

        document.getElementById("errorBox").style.display = "block";

    }
}

</script>
