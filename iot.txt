from machine import Pin
import dht
import time
sensor = dht.DHT22(Pin(15))
while True:
    sensor.measure()
    temp = sensor.temperature()
    hum = sensor.humidity()
    print("Temperature:", temp, "°C")
    print("Humidity:", hum, "%")
    time.sleep(2)
